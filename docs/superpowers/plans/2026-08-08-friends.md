# Friends Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let two real accounts form a mutual, consenting connection by username, and let a club owner/admin quick-add a friend straight into a club without an invite code.

**Architecture:** A `friendships` table (requester/addressee/status) with RLS restricted to the two parties, a thin `src/repos/friendRepo.ts` matching this codebase's existing repo pattern (`if (!supabase) throw`, direct table calls, no local cache — friends data is always fetched live, unlike clubs/players which cache offline), a new `add_friend_to_club` Postgres RPC following the `join_club`/`create_club` security-definer pattern, and two UI surfaces: a Friends screen under Profile, and an "Add friend" entry point on the existing club players-management screen.

**Tech Stack:** Supabase (Postgres + RLS + RPC), React Native / Expo Router, Jest for repo unit tests (this codebase has no UI/component test framework — UI tasks are manually verified by running the app).

## Global Constraints

- Migration numbering: re-check `ls supabase/migrations | tail -3` immediately before writing each migration file — this project has repeatedly collided on migration numbers across concurrent sessions.
- Repo functions guard with `if (!supabase) throw new Error("Sign in to ...")` as the first line, matching every existing repo (`clubRepo.ts`, `sessionRepo.ts`, etc.).
- No `Co-Authored-By` trailer on commits (project convention, `CLAUDE.md`).
- Friendships require both accounts be real (non-anonymous) — enforced by the insert RLS policy checking `is_anonymous` is false, same guard the `profiles` table already uses.
- No fuzzy search or directory/browse of usernames — add-friend is exact-match lookup only, per the usernames spec's anti-enumeration stance.
- Spec deviation (resolved during planning, not a scope change): the spec's phrase "club's manage-players screen (where `addGuestPlayer` currently lives)" was imprecise — `addGuestPlayer` is actually only called from the tournament entrant-add flow (`app/(app)/tournaments/new.tsx`). The real club-roster management screen is `app/(app)/clubs/players/[clubId].tsx`, reached via "Manage players" on the club detail screen. That is the correct integration point and what Task 6 uses.

- [ ] **Step 1: Write the migration**

```sql
-- 0032_friendships.sql
-- A mutual, consenting connection between two real accounts, requested by
-- username and accepted before either side can see the other as a friend.

create table if not exists public.friendships (
  requester_id uuid not null references public.profiles (id) on delete cascade,
  addressee_id uuid not null references public.profiles (id) on delete cascade,
  status text not null default 'pending' check (status in ('pending', 'accepted')),
  created_at timestamptz not null default now(),
  responded_at timestamptz,
  primary key (requester_id, addressee_id),
  check (requester_id <> addressee_id)
);

-- One relationship per pair regardless of direction: blocks A->B and B->A
-- both existing at once (simultaneous requests, or re-requesting someone
-- who already has a pending/accepted row the other way).
create unique index if not exists friendships_unordered_pair
  on public.friendships (least(requester_id, addressee_id), greatest(requester_id, addressee_id));

alter table public.friendships enable row level security;

create policy "friendships are readable by either party"
  on public.friendships for select to authenticated
  using (auth.uid() = requester_id or auth.uid() = addressee_id);

create policy "a user can request a friendship as themselves"
  on public.friendships for insert to authenticated
  with check (
    auth.uid() = requester_id
    and coalesce((auth.jwt()->>'is_anonymous')::boolean, false) = false
  );

create policy "either party can update status"
  on public.friendships for update to authenticated
  using (auth.uid() = requester_id or auth.uid() = addressee_id);

create policy "either party can delete the friendship"
  on public.friendships for delete to authenticated
  using (auth.uid() = requester_id or auth.uid() = addressee_id);
```

```sql
-- 0033_add_friend_to_club.sql
-- Lets a club owner/admin add an accepted friend straight into the club,
-- skipping the invite-code flow. Mirrors join_club/create_club's pattern of
-- inserting both club_members and club_players in one call.

create or replace function public.add_friend_to_club(p_club_id uuid, p_friend_id uuid)
returns void
language plpgsql
security definer
set search_path = public
as $$
declare
  v_role text;
  v_are_friends boolean;
begin
  select role into v_role from club_members
    where club_id = p_club_id and profile_id = auth.uid();
  if v_role is null or v_role not in ('owner', 'admin') then
    raise exception 'Not authorized';
  end if;

  select exists (
    select 1 from friendships
    where status = 'accepted'
      and ((requester_id = auth.uid() and addressee_id = p_friend_id)
        or (requester_id = p_friend_id and addressee_id = auth.uid()))
  ) into v_are_friends;
  if not v_are_friends then
    raise exception 'Not friends with that player';
  end if;

  insert into club_members (club_id, profile_id, role)
  values (p_club_id, p_friend_id, 'member')
  on conflict do nothing;

  insert into club_players (club_id, profile_id, name)
  select p_club_id, p_friend_id, coalesce(p.display_name, 'Player')
  from profiles p where p.id = p_friend_id
  on conflict (club_id, profile_id) do nothing;
end;
$$;

grant execute on function public.add_friend_to_club(uuid, uuid) to authenticated;
```

```typescript
// src/repos/friendRepo.ts
import { supabase } from "../lib/supabase";

export interface Friend {
  id: string;
  displayName: string;
  avatarUrl: string | null;
}

export interface FriendRequestRow {
  profileId: string;
  displayName: string;
  avatarUrl: string | null;
}

async function currentUserId(): Promise<string> {
  const { data } = await supabase!.auth.getUser();
  const uid = data.user?.id;
  if (!uid) throw new Error("Sign in to use friends");
  return uid;
}

async function profilesById(ids: string[]): Promise<Map<string, { displayName: string; avatarUrl: string | null }>> {
  if (ids.length === 0) return new Map();
  const { data, error } = await supabase!
    .from("profiles")
    .select("id, display_name, avatar_url")
    .in("id", ids);
  if (error) throw new Error(error.message);
  return new Map(
    (data ?? []).map((p) => [p.id, { displayName: p.display_name ?? "Player", avatarUrl: p.avatar_url }])
  );
}

/** Sends a friend request by exact username match. Throws if the username doesn't exist, is your own, or a request/friendship already exists in either direction. */
export async function sendFriendRequest(username: string): Promise<void> {
  if (!supabase) throw new Error("Sign in to add a friend");
  const uid = await currentUserId();
  const trimmed = username.trim().toLowerCase();
  const { data: target, error: lookupError } = await supabase
    .from("profiles")
    .select("id")
    .eq("username", trimmed)
    .maybeSingle();
  if (lookupError) throw new Error(lookupError.message);
  if (!target) throw new Error(`No player found with username "${trimmed}"`);
  if (target.id === uid) throw new Error("You can't friend yourself");
  const { error } = await supabase.from("friendships").insert({ requester_id: uid, addressee_id: target.id });
  if (error) {
    if (error.code === "23505") throw new Error("Already friends or request already pending");
    throw new Error(error.message);
  }
}

/** Accept or decline an incoming request. Only the addressee may call this. */
export async function respondToRequest(requesterId: string, accept: boolean): Promise<void> {
  if (!supabase) throw new Error("Sign in to manage friend requests");
  const uid = await currentUserId();
  if (accept) {
    const { error } = await supabase
      .from("friendships")
      .update({ status: "accepted", responded_at: new Date().toISOString() })
      .eq("requester_id", requesterId)
      .eq("addressee_id", uid);
    if (error) throw new Error(error.message);
  } else {
    const { error } = await supabase
      .from("friendships")
      .delete()
      .eq("requester_id", requesterId)
      .eq("addressee_id", uid);
    if (error) throw new Error(error.message);
  }
}

/** Cancels a request you sent that hasn't been responded to yet. */
export async function cancelRequest(addresseeId: string): Promise<void> {
  if (!supabase) throw new Error("Sign in to manage friend requests");
  const uid = await currentUserId();
  const { error } = await supabase
    .from("friendships")
    .delete()
    .eq("requester_id", uid)
    .eq("addressee_id", addresseeId);
  if (error) throw new Error(error.message);
}

/** Removes an accepted friendship. Either party may call this. */
export async function removeFriend(friendId: string): Promise<void> {
  if (!supabase) throw new Error("Sign in to manage friends");
  const uid = await currentUserId();
  const { error } = await supabase
    .from("friendships")
    .delete()
    .or(`and(requester_id.eq.${uid},addressee_id.eq.${friendId}),and(requester_id.eq.${friendId},addressee_id.eq.${uid})`);
  if (error) throw new Error(error.message);
}

export async function listFriends(): Promise<Friend[]> {
  if (!supabase) return [];
  const uid = await currentUserId();
  const { data, error } = await supabase
    .from("friendships")
    .select("requester_id, addressee_id")
    .eq("status", "accepted")
    .or(`requester_id.eq.${uid},addressee_id.eq.${uid}`);
  if (error) throw new Error(error.message);
  const otherIds = (data ?? []).map((r) => (r.requester_id === uid ? r.addressee_id : r.requester_id));
  const profiles = await profilesById(otherIds);
  return otherIds.map((id) => ({ id, ...(profiles.get(id) ?? { displayName: "Player", avatarUrl: null }) }));
}

export async function listIncomingRequests(): Promise<FriendRequestRow[]> {
  if (!supabase) return [];
  const uid = await currentUserId();
  const { data, error } = await supabase
    .from("friendships")
    .select("requester_id")
    .eq("status", "pending")
    .eq("addressee_id", uid);
  if (error) throw new Error(error.message);
  const ids = (data ?? []).map((r) => r.requester_id);
  const profiles = await profilesById(ids);
  return ids.map((id) => ({ profileId: id, ...(profiles.get(id) ?? { displayName: "Player", avatarUrl: null }) }));
}

export async function listOutgoingRequests(): Promise<FriendRequestRow[]> {
  if (!supabase) return [];
  const uid = await currentUserId();
  const { data, error } = await supabase
    .from("friendships")
    .select("addressee_id")
    .eq("status", "pending")
    .eq("requester_id", uid);
  if (error) throw new Error(error.message);
  const ids = (data ?? []).map((r) => r.addressee_id);
  const profiles = await profilesById(ids);
  return ids.map((id) => ({ profileId: id, ...(profiles.get(id) ?? { displayName: "Player", avatarUrl: null }) }));
}
```

```tsx
      await sendFriendRequest(trimmed);
      setUsername("");
      toast(`Friend request sent to ${trimmed}`, "success");
      await reload();
    } catch (e) {
      toast((e as Error).message || "Could not send request", "error");
    } finally {
      setSending(false);
    }
  };

  const onAccept = async (profileId: string) => {
    setBusyId(profileId);
    try {
      await respondToRequest(profileId, true);
      await reload();
    } catch (e) {
      toast((e as Error).message || "Could not accept request", "error");
    } finally {
      setBusyId(null);
    }
  };

    setBusyId(profileId);
    try {
      await respondToRequest(profileId, false);
      await reload();
    } catch (e) {
      toast((e as Error).message || "Could not decline request", "error");
    } finally {
      setBusyId(null);
    }
  };

  const onCancel = async (profileId: string) => {
    setBusyId(profileId);
    try {
      await cancelRequest(profileId);
      await reload();
    } catch (e) {
      toast((e as Error).message || "Could not cancel request", "error");
    } finally {
      setBusyId(null);
    }
  };

  const doRemove = async () => {
    const friend = pendingRemove;
    setPendingRemove(null);
    if (!friend) return;
    setBusyId(friend.id);
    try {
      await removeFriend(friend.id);
      setFriends((cur) => cur.filter((f) => f.id !== friend.id));
      toast(`${friend.displayName} removed`, "success");
    } catch (e) {
      toast((e as Error).message || "Could not remove friend", "error");
    } finally {
      setBusyId(null);
    }
  };

  if (loading) return <View style={s.center}><ActivityIndicator color={C.red} /></View>;

  return (
    <>
      <FlatList
        style={s.root}
        contentContainerStyle={s.content}
        data={friends}
        keyExtractor={(f) => f.id}
        ListHeaderComponent={
          <>
            <View style={s.addRow}>
              <TextInput
                style={s.input}
                placeholder="Username"
                placeholderTextColor={C.text3}
                autoCapitalize="none"
                autoCorrect={false}
                value={username}
                onChangeText={setUsername}
                onSubmitEditing={onAdd}
              />
              <PressableScale style={[s.addBtn, sending && s.off]} disabled={sending} onPress={onAdd}>
                <Text style={s.addBtnText}>Add</Text>
              </PressableScale>
            </View>

            {incoming.length > 0 && (
              <>
                <Text style={s.sectionTitle}>Requests</Text>
                {incoming.map((r) => (
                  <View key={r.profileId} style={s.row}>
                    <Text style={s.name} numberOfLines={1}>{r.displayName}</Text>
                    <PressableScale
                      style={[s.chip, busyId === r.profileId && s.off]}
                      disabled={busyId === r.profileId}
                      onPress={() => onAccept(r.profileId)}
                    >
                      <Text style={s.chipText}>Accept</Text>
                    </PressableScale>
                    <PressableScale
                      style={[s.chip, s.chipMuted, busyId === r.profileId && s.off]}
                      disabled={busyId === r.profileId}
                      onPress={() => onDecline(r.profileId)}

                      <Text style={s.chipTextMuted}>Decline</Text>
                    </PressableScale>
                  </View>

            {outgoing.length > 0 && (

                    >
                      <Text style={s.chipTextMuted}>Cancel</Text>
                    </PressableScale>
                  </View>
                ))}
              </>
            )}

            <Text style={s.sectionTitle}>Friends</Text>
          </>
        }
        ListEmptyComponent={<Text style={s.empty}>No friends yet. Add one by username above.</Text>}

    </>
  );
}
```

```tsx
import React, { useCallback, useState } from "react";
import { ActivityIndicator, FlatList, StyleSheet, Text, View } from "react-native";
import { useFocusEffect, useLocalSearchParams } from "expo-router";
import { addFriendToClub, getClub, listPlayers, removePlayer, type ClubPlayer } from "../../../../src/repos/clubRepo";
import { listFriends, type Friend } from "../../../../src/repos/friendRepo";
import { toast } from "../../../../src/lib/toast";
import Sheet from "../../../../src/components/Sheet";
import Icon from "../../../../src/components/Icon";
import PressableScale from "../../../../src/components/PressableScale";
import { C, F } from "../../../../src/theme";

export default function ClubPlayers() {
  const { clubId } = useLocalSearchParams<{ clubId: string }>();
  const [players, setPlayers] = useState<ClubPlayer[]>([]);
  const [isAdmin, setIsAdmin] = useState(false);
  const [loading, setLoading] = useState(true);
  const [busyId, setBusyId] = useState<string | null>(null);
  const [pending, setPending] = useState<ClubPlayer | null>(null);
  const [pickerOpen, setPickerOpen] = useState(false);
  const [addableFriends, setAddableFriends] = useState<Friend[]>([]);
  const [addingId, setAddingId] = useState<string | null>(null);

  const reload = useCallback(async () => {
    if (!clubId) return;
    try {
      const [list, club] = await Promise.all([listPlayers(clubId), getClub(clubId)]);
      setPlayers(list);
      setIsAdmin(club?.role === "owner" || club?.role === "admin");
    } catch (e) {
      toast((e as Error).message || "Could not load players", "error");
    } finally {
      setLoading(false);
    }
  }, [clubId]);
  useFocusEffect(useCallback(() => { void reload(); }, [reload]));

  const doRemove = async () => {
    const player = pending;
    setPending(null);
    if (!player) return;
    setBusyId(player.id);
    try {
      await removePlayer(player.id);
      setPlayers((cur) => cur.filter((p) => p.id !== player.id));
      toast(`${player.name} removed`, "success");
    } catch (e) {
      toast((e as Error).message || "Could not remove that player", "error");
    } finally {
      setBusyId(null);
    }
  };

  const openPicker = async () => {
    try {
      const friends = await listFriends();
      const existingIds = new Set(players.map((p) => p.profile_id).filter(Boolean));
      setAddableFriends(friends.filter((f) => !existingIds.has(f.id)));
      setPickerOpen(true);
    } catch (e) {
      toast((e as Error).message || "Could not load friends", "error");
    }
  };

  const onAddFriend = async (friend: Friend) => {
    if (!clubId) return;
    setAddingId(friend.id);
    try {
      await addFriendToClub(clubId, friend.id);
      toast(`${friend.displayName} added`, "success");
      setPickerOpen(false);
      await reload();
    } catch (e) {
      toast((e as Error).message || "Could not add that friend", "error");
    } finally {
      setAddingId(null);
    }
  };

  if (loading) return <View style={s.center}><ActivityIndicator color={C.red} /></View>;

  return (
    <>
      <FlatList
        style={s.root}
        contentContainerStyle={s.content}
        data={players}
        keyExtractor={(p) => p.id}
        ListHeaderComponent={
          isAdmin ? (
            <PressableScale style={s.addFriendBtn} onPress={openPicker}>
              <Icon name="person-add-outline" color={C.text} size={18} />
              <Text style={s.addFriendText}>Add friend</Text>
            </PressableScale>
          ) : null
        }
        ListEmptyComponent={<Text style={s.empty}>No players yet.</Text>}
        renderItem={({ item }) => (
          <View style={s.row}>
            <Text style={s.name} numberOfLines={1}>{item.name}</Text>
            {isAdmin ? (
              <PressableScale
                style={[s.remove, busyId === item.id && s.off]}
                disabled={busyId === item.id}
                onPress={() => setPending(item)}
              >
                <Icon name="trash-outline" color={C.text2} size={18} />
              </PressableScale>
            ) : null}
          </View>
        )}
      />
      <Sheet
        visible={pickerOpen}
        title="Add friend to club"
        actions={
          addableFriends.length > 0
            ? [
                ...addableFriends.map((f) => ({
                  label: addingId === f.id ? `Adding ${f.displayName}...` : f.displayName,
                  onPress: () => onAddFriend(f),
                })),
                { label: "Cancel", onPress: () => setPickerOpen(false) },
              ]
            : [{ label: "Cancel", onPress: () => setPickerOpen(false) }]
        }
        message={addableFriends.length === 0 ? "No friends available to add — everyone you're friends with is already in this club." : undefined}
        onClose={() => setPickerOpen(false)}
      />
      <Sheet
        visible={!!pending}
        title="Remove player?"
        message={pending ? `${pending.name} will be removed from the roster. They can't have any tournament history.` : ""}
        actions={[
          { label: "Remove", onPress: doRemove, destructive: true },
          { label: "Cancel", onPress: () => setPending(null) },
        ]}
        onClose={() => setPending(null)}
      />
    </>
  );
}
```
